      /// Reload configuration: read the conf
ig file, diff against current, and
      /// apply hot-reloadable actions. Retur
ns the reload plan for API response.
      pub fn reload_config(&self) -> Result<c
rate::config_reload::ReloadPlan, String> {
          use crate::config_reload::{
              build_reload_plan, should_apply
_hot, validate_config_for_reload,
          };
  
          // Read and parse config file (usin
g load_config to process $include directives)
          let config_path = self.config.home_
dir.join("config.toml");
          let new_config = if config_path.exi
sts() {
              crate::config::load_config(Some
(&config_path))
          } else {
              return Err("Config file not fou
nd".to_string());
          };
  
          // Validate new config
          if let Err(errors) = validate_confi
g_for_reload(&new_config) {
              return Err(format!("Validation 
failed: {}", errors.join("; ")));
          }
  
          // Build the reload plan
          let plan = build_reload_plan(&self.

          plan.log_summary();
  
          // Apply hot actions if the reload

          if should_apply_hot(self.config.rel
oad.mode, &plan) {
              self.apply_hot_actions(&plan, &
new_config);
          }
  
          Ok(plan)
      }
